cmake_minimum_required(VERSION 3.21)
project(MacGlassRefraction VERSION 0.1 LANGUAGES CXX)
option(MACGLASS_NESTED_TEST_CAPTURE "Capture only in a controlled, isolated test compositor. NEVER install this build." OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Qml DBus Quick Widgets)
# The distribution's KWinConfig requires ECM just to locate pkg-config dependencies.
# Use the installed SDK directly; do not fake ECM or download/install build packages.
find_package(PkgConfig REQUIRED)
pkg_check_modules(KWIN_DEPS REQUIRED IMPORTED_TARGET epoxy wayland-server libdrm)
find_package(KF6Config REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_path(KWIN_INCLUDE_DIR effect/effect.h PATH_SUFFIXES kwin REQUIRED)
find_library(KWIN_LIBRARY NAMES kwin REQUIRED)
file(STRINGS "${KWIN_INCLUDE_DIR}/config-kwin.h" KWIN_ABI REGEX "^#define KWIN_PLUGIN_VERSION_STRING")
if(NOT KWIN_ABI MATCHES "\"6\\.8\\.0\"")
    message(FATAL_ERROR "This experimental module is audited for KWin 6.8.0 only")
endif()
add_library(KWin::kwin SHARED IMPORTED)
set_target_properties(KWin::kwin PROPERTIES IMPORTED_LOCATION "${KWIN_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${KWIN_INCLUDE_DIR}"
    INTERFACE_LINK_LIBRARIES "Qt6::DBus;Qt6::Quick;Qt6::Widgets;KF6::ConfigCore;KF6::CoreAddons;KF6::WindowSystem;PkgConfig::KWIN_DEPS")
set(shader_identity "")
foreach(shader lens.vert lens.frag motion.frag)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${shader}" shader_hash)
    string(APPEND shader_identity "${shader_hash}")
endforeach()
string(SHA256 shader_identity "${shader_identity}")
set(shader_prefix "/macglass/${shader_identity}")
set(shader_resource "${CMAKE_CURRENT_BINARY_DIR}/macglass_${shader_identity}.qrc")
file(WRITE "${shader_resource}" "<RCC><qresource prefix=\"${shader_prefix}\"><file alias=\"lens.vert\">${CMAKE_CURRENT_SOURCE_DIR}/lens.vert</file><file alias=\"lens.frag\">${CMAKE_CURRENT_SOURCE_DIR}/lens.frag</file><file alias=\"motion.frag\">${CMAKE_CURRENT_SOURCE_DIR}/motion.frag</file></qresource></RCC>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS lens.vert lens.frag motion.frag)
add_library(macglassrefraction MODULE refraction.cpp motion.cpp panelgenie.cpp dockintegration.cpp refraction.h "${shader_resource}")
target_compile_definitions(macglassrefraction PRIVATE MACGLASS_SHADER_PREFIX=":${shader_prefix}/")
set_property(TARGET macglassrefraction APPEND PROPERTY AUTOMOC_MACRO_NAMES KWIN_EFFECT_FACTORY_SUPPORTED_ENABLED)
target_link_libraries(macglassrefraction PRIVATE KWin::kwin Qt6::DBus)
target_link_options(macglassrefraction PRIVATE "LINKER:--no-undefined" "LINKER:-Bsymbolic-functions")
if(MACGLASS_NESTED_TEST_CAPTURE)
    target_compile_definitions(macglassrefraction PRIVATE MACGLASS_NESTED_TEST_CAPTURE)
endif()
set_target_properties(macglassrefraction PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/kwin/effects/plugins")
add_library(macglassbootstrap MODULE bootstrap.cpp)
target_link_libraries(macglassbootstrap PRIVATE KWin::kwin Qt6::Qml)
set_target_properties(macglassbootstrap PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native")
configure_file(bootstrap.qml bootstrap.qml COPYONLY)
configure_file(register.qml register.qml COPYONLY)
configure_file(reload.qml reload.qml COPYONLY)
configure_file(qmldir native/qmldir COPYONLY)
# Intentionally no automatic installation or compositor loading.
